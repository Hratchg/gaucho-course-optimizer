    **Part A — GradeChart (src/components/GradeChart.tsx):**

    Grade distribution bar chart aggregating counts across all quarters for a professor-course pair.

    ```tsx
    import { BarChart, Bar, XAxis, YAxis, ResponsiveContainer, Tooltip } from 'recharts'
    import type { GradeQuarter } from '@/types/api'

    const GRADE_KEYS = [
      'a_plus', 'a', 'a_minus', 'b_plus', 'b', 'b_minus',
      'c_plus', 'c', 'c_minus', 'd_plus', 'd', 'd_minus', 'f',
    ] as const

    const GRADE_LABELS = [
      'A+', 'A', 'A-', 'B+', 'B', 'B-',
      'C+', 'C', 'C-', 'D+', 'D', 'D-', 'F',
    ]

    interface GradeChartProps {
      quarters: GradeQuarter[]
    }

    export function GradeChart({ quarters }: GradeChartProps) {
      const data = GRADE_LABELS.map((label, i) => ({
        grade: label,
        count: quarters.reduce((sum, q) => sum + (q[GRADE_KEYS[i]] ?? 0), 0),
      }))

      return (
        <ResponsiveContainer width="100%" height={300}>
          <BarChart data={data} margin={{ top: 8, right: 8, bottom: 8, left: 8 }}>
            <XAxis dataKey="grade" tick={{ fontSize: 12 }} />
            <YAxis tick={{ fontSize: 12 }} />
            <Tooltip />
            <Bar dataKey="count" fill="#6366f1" />
          </BarChart>
        </ResponsiveContainer>
      )
    }
    ```

    Key details:
    - 13 letter grades on X-axis (A+ through F) — per CONTEXT.md Specifics
    - Student count on Y-axis (aggregate across all quarters)
    - Bar fill color: `#6366f1` (indigo) — per UI-SPEC Color section, chart-only color
    - ResponsiveContainer with `width="100%"` and explicit `height={300}` (Pitfall 4)
    - Tick font size 12px per Typography Label role

    **Part B — GpaTrendChart (src/components/GpaTrendChart.tsx):**

    GPA trend line chart showing average GPA per quarter over time.

    ```tsx
    import { LineChart, Line, XAxis, YAxis, ResponsiveContainer, Tooltip } from 'recharts'
    import type { GradeQuarter } from '@/types/api'

    interface GpaTrendChartProps {
      quarters: GradeQuarter[]
    }

    export function GpaTrendChart({ quarters }: GpaTrendChartProps) {
      const data = quarters
        .filter((q) => q.avg_gpa != null)
        .map((q) => ({ quarter: q.quarter, avg_gpa: q.avg_gpa }))

      if (data.length === 0) return <p className="text-sm text-muted-foreground">No GPA trend data available</p>

      return (
        <ResponsiveContainer width="100%" height={250}>
          <LineChart data={data}>
            <XAxis dataKey="quarter" tick={{ fontSize: 12 }} />
            <YAxis domain={[0, 4.0]} tick={{ fontSize: 12 }} />
            <Tooltip />
            <Line type="monotone" dataKey="avg_gpa" stroke="#10b981" dot={false} />
          </LineChart>
        </ResponsiveContainer>
      )
    }
    ```

    Key details:
    - Y-axis domain fixed at [0, 4.0] — per Streamlit prototype `yaxis_range=[0, 4.0]`
    - Line stroke color: `#10b981` (emerald) — per UI-SPEC Color section, chart-only color
    - Filter out quarters with null avg_gpa before rendering
    - Explicit `height={250}` on ResponsiveContainer (Pitfall 4)
    - `dot={false}` for clean line appearance

    function ExpandedComments({ professorId }: { professorId: number }) {
      const { data: comments, isLoading } = useProfessorComments(professorId)

      if (isLoading) return <p className="text-sm text-muted-foreground">Loading comments...</p>

        <div>
          <h4 className="mb-2 text-sm font-semibold">Recent Comments</h4>
          <div className="space-y-3">
            {comments.map((comment, i) => (
              <div key={i} className="border-b pb-3 last:border-b-0">
                <div className="flex items-center justify-between">
                  <span className="text-xs text-muted-foreground">
                    {comment.created_at ?? 'Unknown date'}
                  </span>
                  <SentimentBadge score={comment.sentiment_score} />
                </div>
                <p className="mt-1 text-sm leading-relaxed">
                  {comment.text ?? 'No comment text'}
                </p>
              </div>
            ))}
          </div>
        </div>
      )
    }
